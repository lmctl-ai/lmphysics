cmake_minimum_required(VERSION 3.24)
project(lmphysics VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMPHYSICS_FULL_STATIC "Link the complete executable statically" ON)

foreach(dep eigen chrono)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dep}")
    message(FATAL_ERROR "third_party/${dep} missing: run scripts/sync-deps.sh")
  endif()
endforeach()

# Embed schema and notices.
function(embed_file var file ns name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
embed_file(SCHEMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmphysics.dsl.json" lmphysics embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmphysics { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" lmphysics embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmphysics { ${NOTICES_SRC} }\n")

include(cmake/Chrono.cmake)

add_executable(lmphysics cpp/main.cpp)
target_compile_definitions(lmphysics PRIVATE LMPHYSICS_VERSION="${PROJECT_VERSION}")
target_include_directories(lmphysics PRIVATE cpp "${GEN}")
target_link_libraries(lmphysics PRIVATE lmphysics_chrono)
target_compile_options(lmphysics PRIVATE -Wall -Wextra -Wno-unused-parameter)
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmphysics PRIVATE -static-libstdc++ -static-libgcc)
  if(LMPHYSICS_FULL_STATIC)
    target_link_options(lmphysics PRIVATE -static)
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(lmphysics PRIVATE -s)
  endif()
endif()
